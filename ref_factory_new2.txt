var ref = function (d) {
  return {
    id: d.id,
    ref: true,
    views: d.views,
    tag: d.tag,
    trad: d.trad,
    num: d.num,
    rows: d.rows,
    source: d.source,
    title: d.title,
    note: d.note,
    lede: d.lede,
    html: function () {
      return '<div class="nq-ctl" data-n="rows" style="display:grid;gap:8px"></div>' +
        '<div class="nq-note" data-n="status" style="margin-top:10px"></div>';
    },
    back: function () {
      return "<p>" + d.lede + "</p>" + entryBlock(d.rows);
    },
    update: function (n, D) {
      var target = null;
      if (d.id === "ref_core" || d.id === "ref_sep" || d.id === "ref_loshu") {
        if (D.letters.length) target = masterReduce(D.letters.reduce(function (a, ch) { return a + pythValue(ch.charCodeAt(0)); }, 0));
        else target = masterReduce(D.digits.reduce(function (a, b) { return a + b; }, 0));
      } else if (d.id === "ref_chal") {
        if (D.letters.length) target = masterReduce(D.letters.reduce(function (a, ch) { return a + chaldeanValue(ch.charCodeAt(0)); }, 0));
        else target = masterReduce(D.digits.reduce(function (a, b) { return a + b; }, 0));
      } else if (d.id === "ref_life") {
        var groups = [];
        if (/[-/.]/.test(D.raw)) groups = D.raw.split(/[-/.]+/).map(function (g) { return g.replace(/\D/g, ""); }).filter(Boolean);
        else if (/^\d{8}$/.test(D.raw.trim())) { var s = D.raw.trim(); groups = [s.slice(0, 4), s.slice(4, 6), s.slice(6, 8)]; }
        else if (/^\d{6}$/.test(D.raw.trim())) { var s = D.raw.trim(); groups = [s.slice(0, 2), s.slice(2, 4), s.slice(4, 6)]; }
        if (groups.length >= 2) {
          var comps = groups.map(function (g) { return masterReduce(parseInt(g, 10)); });
          target = masterReduce(comps.reduce(function (a, b) { return a + b; }, 0));
        } else target = masterReduce(D.digits.reduce(function (a, b) { return a + b; }, 0));
      }
      n.val.textContent = "\u2014";
      n.n.textContent = d.rows.length;
      if (n.rows) {
        var html = d.rows.map(function (pair) {
          var match = target !== null && (String(pair[0]) === String(target) || pair[0].indexOf(String(target)) >= 0);
          var pad = match ? "padding:10px 12px;border:1px solid var(--nq-gold);background:color-mix(in srgb,var(--nq-gold) 14%,transparent);color:var(--nq-gold);font-weight:700;" : "padding:10px 12px;border:1px solid var(--nq-faint);background:transparent;";
          return '<div style="display:flex;gap:14px;align-items:flex-start;border-radius:8px;' + pad + '">' +
            '<b style="font-family:var(--nq-mono);font-size:15px;min-width:4.6em;align-self:flex-start">' + pair[0] + '</b>' +
            '<span style="flex:1;line-height:1.55">' + pair[1] + '</span></div>';
        }).join("");
        n.rows.innerHTML = html;
      }
      if (n.status) {
        if (target === null) n.status.textContent = "";
        else {
          var found = d.rows.some(function (pair) { return String(pair[0]) === String(target) || pair[0].indexOf(String(target)) >= 0; });
          n.status.textContent = found ? ("Your current input highlights entry " + target + ".") : ("Your current input reduces to " + target + "; the closest set above carries its meaning.");
        }
      }
      if (D) D.copy = d.title;
    }
  };
};
